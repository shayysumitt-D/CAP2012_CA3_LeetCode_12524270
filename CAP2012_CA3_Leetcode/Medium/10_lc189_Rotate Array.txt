class Solution {
public:
    void rotate(vector<int>& nums, int k) {
        int n = nums.size();
        reverse(nums, 0, n - 1);
        reverse(nums, 0, k % n - 1);
        reverse(nums, k % n, n - 1);
        
    }

    void reverse(vector<int>& nums, int s, int e){
        while(s < e) {
            swap(nums[s], nums[e]);
            s++; 
            e--;
        }
    }
};
